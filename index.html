구글 스프레드시트 (Apps Script) 연동 설정

📌 구글 스프레드시트와 어떻게 연결하나요?

  1. Google Drive에서 새 스프레드시트를 만듭니다.
  2. 상단 메뉴의 [확장 프로그램] → [Apps Script]를 클릭합니다.
  3. 아래의 [Apps Script 예시 코드]를 붙여넣고 [배포] → [새 배포]를 선택합니다.
  4. 유형: 웹 앱, 액세스 권한: 모든 사용자로 배포 후 생성된 웹 앱 URL을 아래에 입력하세요.

입력한 URL은 브라우저(localStorage)에 안전하게 자동 저장됩니다.

📋 복사해서 쓸 수 있는 Apps Script 백엔드 코드 보기 
// Google Apps Script (Code.gs)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('역할설정') || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  // 1행 헤더(학생이름, 역할이름, 역할설명)
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) {
      roles.push({
        title: data[i][1],
        desc: data[i][2] || ''
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('배정결과');
  if (!sheet) {
    sheet = ss.insertSheet('배정결과');
    sheet.appendRow(['배정일시', '학급명', '번호', '학생이름', '배정역할', '역할설명']);
  }
  var payload = JSON.parse(e.postData.contents);
  var timestamp = new Date();
  payload.assignments.forEach(function(item, idx) {
    sheet.appendRow([timestamp, payload.className, idx + 1, item.student, item.role, item.desc]);
  });
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    savedCount: payload.assignments.length
  })).setMimeType(ContentService.MimeType.JSON);
}
